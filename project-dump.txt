## 🚀 Project Structure

Inside of your Astro project, you'll see the following folders and files:

```text
/
├── public/
│   └── favicon.svg
├── src
│   ├── assets
│   │   └── astro.svg
│   ├── components
│   │   └── Welcome.astro
│   ├── layouts
│   │   └── Layout.astro
│   └── pages
│       └── index.astro
└── package.json
```

To learn more about the folder structure of an Astro project, refer to [our guide on project structure](https://docs.astro.build/en/basics/project-structure/).

## 🧞 Commands

All commands are run from the root of the project, from a terminal:

| Command                   | Action                                           |
| :------------------------ | :----------------------------------------------- |
| `npm install`             | Installs dependencies                            |
| `npm run dev`             | Starts local dev server at `localhost:4321`      |
| `npm run build`           | Build your production site to `./dist/`          |
| `npm run preview`         | Preview your build locally, before deploying     |
| `npm run astro ...`       | Run CLI commands like `astro add`, `astro check` |
| `npm run astro -- --help` | Get help using the Astro CLI                     |

## 👀 Want to learn more?

Feel free to check [our documentation](https://docs.astro.build) or jump into our [Discord server](https://astro.build/chat).

import { glob } from "astro/loaders";
// 从 `astro:content` 导入工具函数
import { defineCollection } from "astro:content";
// 导入 Zod
import { z } from "astro/zod";

// 为每个集合定义一个 `loader` 和 `schema`
const blog = defineCollection({
  loader: glob({ pattern: "**/[^_]*.mdx", base: "./src/blog" }),
  schema: z.object({
    title: z.string(),
    pubDate: z.date(),
    description: z.string(),
    author: z.string(),
    image: z.object({
      url: z.string(),
      alt: z.string(),
    }),
    tags: z.array(z.string()),
    primaryTag: z.string().optional(), // 主标签，用于面包屑导航，必须是 tags 数组中的一个值

    canonicalURL: z.string().optional(), // 可选，手动指定规范链接，防止重复内容被搜索引擎降权
  }).refine(
    (data) => !data.primaryTag || data.tags.includes(data.primaryTag),
    { message: "primaryTag 必须是 tags 数组中已有的一个值" } // 构建时自动校验，防止输入错误导致面包屑 404
  ),
});

// 导出一个单独的 `collections` 对象用以注册你的集合（们）
export const collections = { blog };

 * 面包屑导航组件：同时输出可见的导航路径 + JSON-LD 结构化数据
 *
 * ═══════════════════════════════════════════════════
 * 用法一：直接传完整数组（适合简单固定页面）
 * ═══════════════════════════════════════════════════
 *
 * 关于页面 about.astro：
 *   <Breadcrumb items={[
 *     { name: "首页", url: "/" },
 *     { name: "关于", url: "/about/" },
 *   ]} />
 *   效果：首页 › 关于
 *
 * 博客汇总页 blog/index.astro：
 *   <Breadcrumb items={[
 *     { name: "首页", url: "/" },
 *     { name: "博客", url: "/blog/" },
 *   ]} />
 *   效果：首页 › 博客
 *
 * 标签列表页 tags/index.astro：
 *   <Breadcrumb items={[
 *     { name: "首页", url: "/" },
 *     { name: "标签", url: "/tags/" },
 *   ]} />
 *   效果：首页 › 标签
 *
 * 单个标签页 tags/[tag].astro：
 * ---
 *   const { tag } = Astro.params;
 * ---
 *   <Breadcrumb items={[
 *     { name: "首页", url: "/" },
 *     { name: "标签", url: "/tags/" },
 *     { name: tag, url: `/tags/${tag}/` },
 *   ]} />
 *   效果：首页 › 标签 › aluminum
 *
 * ═══════════════════════════════════════════════════
 * 用法二：传结构化数据，组件自动组装
 *         （适合有可选中间分类层的内容详情页）
 * ═══════════════════════════════════════════════════
 *
 * 博客文章页 blog/[...slug].astro（有分类）：
 *   <Breadcrumb
 *     root={{ name: "博客", url: "/blog/" }}
 *     category={post.data.category
 *       ? { name: post.data.category, url: `/tags/${post.data.category}/` }
 *       : undefined
 *     }
 *     current={post.data.title}
 *     currentUrl={`/blog/${post.id}/`}
 *   />
 *   效果（有分类）：首页 › 博客 › 铝型材 › 文章标题
 *   效果（无分类）：首页 › 博客 › 文章标题

 *     category={{ name: "铝型材", url: "/products/aluminum/" }}
 *     current={product.data.name}
 *     currentUrl={`/products/${product.id}/`}
 *   />
 *   效果：首页 › 产品 › 铝型材 › 产品名称
 *
 * ═══════════════════════════════════════════════════
 * 选哪种用法？
 *   有可选中间层（如分类）→ 用法二
 *   路径固定或只有一个动态值 → 用法一

interface BreadcrumbItem {
  name: string;
  url: string;
}

interface Props {
  // 用法一：直接传完整数组
  items?: BreadcrumbItem[];

  // 用法二：传结构化数据，组件自动组装
  root?: BreadcrumbItem;      // 根节点，如 { name: "博客", url: "/blog/" }
  primaryTag?: BreadcrumbItem;  // 可选的中间分类层
  current?: string;           // 当前页名称（文章标题 / 产品名）
  currentUrl?: string;        // 当前页 URL
}

const { items, root, primaryTag, current, currentUrl } = Astro.props;

// 用法一：直接用传入的数组
// 用法二：自动组装 首页 → root → category(可选) → current
const resolvedItems: BreadcrumbItem[] = items ?? [
  { name: "首页", url: "/" },
  ...(root        ? [root]                              : []),
  ...(primaryTag  ? [primaryTag]                        : []),
  ...(current && currentUrl ? [{ name: current, url: currentUrl }] : []),
];

const jsonLd = {
  "@context": "https://schema.org",
  "@type": "BreadcrumbList",
  "itemListElement": resolvedItems.map((item, index) => ({
    "@type": "ListItem",
    "position": index + 1,
    "name": item.name,
    "item": new URL(item.url, Astro.site).href,
  })),
};
